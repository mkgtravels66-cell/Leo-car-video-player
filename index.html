<script>
    // Tab Switching Function
    function switchTab(tabId, btn) {
        document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
        document.getElementById(tabId).classList.add('active');

        document.querySelectorAll('sidebar button').forEach(b => b.classList.remove('active'));
        btn.classList.add('active');
    }

    // YouTube Video Loader logic
    function loadYouTube() {
        const query = document.getElementById('ytSearchQuery').value.trim();
        const screen = document.getElementById('ytScreen');
        
        if (!query) {
            alert("Please enter a valid YouTube link or keyword!");
            return;
        }

        // Check if input is a full youtube URL or watch ID
        let videoId = "jfKfPfyJRdk"; // Default lofi chill beats as fallback
        if (query.includes("youtu.be/")) {
            videoId = query.split("youtu.be/")[1].split("?")[0];
        } else if (query.includes("watch?v=")) {
            videoId = query.split("watch?v=")[1].split("&")[0];
        }

        screen.innerHTML = `<iframe src="https://www.youtube.com/embed/${videoId}?autoplay=1&enablejsapi=1" allow="autoplay; encrypted-media" allowfullscreen></iframe>`;
    }

    // Spotify Link Loader
    function loadSpotify() {
        const url = document.getElementById('spotifyUrl').value.trim();
        const screen = document.getElementById('spotifyScreen');
        if (!url) {
            alert("Please paste a Spotify link!");
            return;
        }
        // Convert regular spotify track link to embed link format
        let embedUrl = url;
        if(url.includes("track/")) {
            let trackId = url.split("track/")[1].split("?")[0];
            embedUrl = `https://open.spotify.com/embed/track/${trackId}`;
        }
        screen.innerHTML = `<iframe src="${embedUrl}?utm_source=generator&theme=0" allow="autoplay; clipboard-write; encrypted-media; fullscreen; picture-in-picture"></iframe>`;
    }

    // Custom Web URL Loader (Fermata / AAAD WebView behavior)
    function loadCustomUrl() {
        let url = document.getElementById('customUrl').value.trim();
        const screen = document.getElementById('browserScreen');
        if(!url.startsWith('http')) {
            url = 'https://' + url;
        }
        screen.innerHTML = `<iframe src="${url}" sandbox="allow-scripts allow-same-origin allow-forms"></iframe>`;
    }
